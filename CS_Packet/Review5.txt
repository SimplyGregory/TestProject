def factorial(n):
    result = 1
    for i in range(1, n + 1):
        result = result * i
    return result

n = int(input("What is your number?"))
print(factorial(n))

def odds(n):
    for i in range(1, n + 1):
        if i % 2 != 0:
            print(i)

n = int(input("What is your number?"))
odds(n)

def perfect(n):
    total = 0
    for i in range(1, n):
        if n % i == 0:
            total += i
    if total == n:
        return True
    else:
        return False

n = int(input("What is your number?"))
print(perfect(n))

def palindrome(word):
    word = word.lower()
    if word == word[::-1]:
        return True
    else:
        return False

word = str(input("What is your word?"))
print(palindrome(word))

if gender == "female":
    retire_age = 62
else:
    retire_age = 65

if age >= retire_age:
    print(f"{name}, you are eligible for retirement!")
else:
    print(f"{name}, you need to wait {retire_age - age} years to get your pension.")

choice = str(input("Convert to Celsius or Fahrenheit? (C/F)")).lower()
temp = float(input("What is your temperature?"))

if choice == "c":
    print(f"{temp} Fahrenheit is {(temp - 32) * 5 / 9} Celsius")
elif choice == "f":
    print(f"{temp} Celsius is {(temp * 9 / 5) + 32} Fahrenheit")
else:
    print("That is not a choice!")
